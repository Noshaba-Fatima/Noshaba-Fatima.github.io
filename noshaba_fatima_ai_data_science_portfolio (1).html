Orchestration & MCP 
from pydantic import BaseModel, ValidationError
from langgraph.graph import StateGraph, END

class ToolCall(BaseModel):
    server: str
    method: str
    args: dict

def dispatch_mcp_tool(payload: dict, mcp_client):
    # Typed schema guardrails + deterministic fallback
    try:
        call = ToolCall.model_validate(payload)
        return mcp_client.invoke(call.server, call.method, call.args)
    except ValidationError as err:
        return {"status": "fallback", "reason": err.errors()}

def compile_agent(router_fn, tool_node):
    wf = StateGraph(dict)
    wf.add_node("tools", tool_node)
    wf.add_conditional_edges("tools", router_fn, {"done": END})
    return wf.compile()
02_rag_and_evals.py
LLMOps & Testing 
import pytest
from langsmith import traceable  # active roadmap

def recall_at_k(retrieved: list[str], gold: set[str], k: int = 5) -> float:
    hits = len(set(retrieved[:k]) & gold)
    return hits / max(len(gold), 1)

@traceable(run_type="chain", name="rag_component_eval")
def eval_rag_components(retriever, generator, golden_set):
    scores = []
    for case in golden_set:
        ctx = retriever.search(case["query"], top_k=5)
        ans = generator.score(case["query"], ctx)
        scores.append((recall_at_k(ctx, case["gold"]), ans.faithful))
    return scores

def test_retriever_floor():
    assert recall_at_k(["c1", "c4"], {"c1"}, k=2) == 1.0
03_ml_baselines_xai.py
Data Science & XAI 
import shap
from sklearn.linear_model import Ridge
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import cross_val_score

def benchmark_and_explain(X, y):
    # 1. Benchmark linear baseline before tree ensembles
    baseline = Ridge(alpha=1.0)
    forest = RandomForestRegressor(n_estimators=200, random_state=42)

    base_rmse = -cross_val_score(baseline, X, y, scoring="neg_root_mean_squared_error").mean()
    tree_rmse = -cross_val_score(forest, X, y, scoring="neg_root_mean_squared_error").mean()

    # 2. Global + local SHAP feature attribution
    forest.fit(X, y)
    shap_values = shap.TreeExplainer(forest).shap_values(X)
    return {"baseline_rmse": base_rmse, "rf_rmse": tree_rmse, "shap": shap_values}
